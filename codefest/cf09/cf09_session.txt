     Step 2 — include/kalman_hw.h

     #ifndef KALMAN_HW_H
     #define KALMAN_HW_H

     #include <stdint.h>

     /* Kalman accelerator base address in VeeRwolf SoC AXI memory map */
     #define KALMAN_BASE  ((volatile uint32_t *)0x80010000U)

     /* Register byte offsets (8-byte stride, 64-bit IEEE-754 doubles) */
     #define KALMAN_CTRL   0x00U
     #define KALMAN_STAT   0x08U
     #define KALMAN_Z      0x10U
     #define KALMAN_XIN0   0x18U
     #define KALMAN_XIN1   0x20U
     #define KALMAN_XIN2   0x28U
     #define KALMAN_XOUT0  0x30U
     #define KALMAN_XOUT1  0x38U
     #define KALMAN_XOUT2  0x40U
     #define KALMAN_PIN0   0x58U   /* P_in[0,0] */
     #define KALMAN_POUT0  0xA0U   /* P_out[0,0] */
     #define KALMAN_RREG   0xE8U

     /* Write one 64-bit IEEE-754 double to an MMIO register.
      * Uses two 32-bit stores (lo then hi) — generates wstrb byte-enables on
      * the AXI4-Lite bus so each half updates its respective bytes. */
     static inline void kw_write(uint32_t offset, double val) {
         union { double d; uint32_t u[2]; } v;
         v.d = val;
         volatile uint32_t *p = KALMAN_BASE + (offset >> 2);
         p[0] = v.u[0];   /* lo word → bytes [31:0]  */
         p[1] = v.u[1];   /* hi word → bytes [63:32] */
     }

     /* Read one 64-bit IEEE-754 double from an MMIO register. */
     static inline double kw_read(uint32_t offset) {
         union { double d; uint32_t u[2]; } v;
         const volatile uint32_t *p = KALMAN_BASE + (offset >> 2);
         v.u[0] = p[0];
         v.u[1] = p[1];
         return v.d;
     }

     /* ── Public API ─────────────────────────────────────────────────────── */

     /* Function 1: Write the measurement noise covariance R to R_REG (0xE8).
      * Call once before the update loop. */
     static inline void kalman_hw_set_r(double r) {
         kw_write(KALMAN_RREG, r);
     }

     /* Function 2: Load z / x_in / P_in, fire the measurement-correction 
     update,
      * poll until done, then read back x_out and P_out.
      *
      * x_in  / x_out : arrays of 3 doubles (state vector)
      * P_in  / P_out : arrays of 9 doubles (3×3 covariance, row-major)
      * Returns 0 on success, -1 if STAT.busy never clears (hung accelerator).
      */
     static inline int kalman_hw_update(double  z,
                                        const double x_in[3],
                                        const double P_in[9],
                                        double  x_out[3],
                                        double  P_out[9]) {
         /* 1. Write z */
         kw_write(KALMAN_Z, z);

         /* 2. Write x_in[0:2] at 0x18, 0x20, 0x28 */
         kw_write(KALMAN_XIN0, x_in[0]);
         kw_write(KALMAN_XIN1, x_in[1]);
         kw_write(KALMAN_XIN2, x_in[2]);

         /* 3. Write P_in[0:8] at 0x58..0x98 (8-byte stride) */
         for (int i = 0; i < 9; i++)
             kw_write(KALMAN_PIN0 + (uint32_t)(i * 8), P_in[i]);

         /* 4. Fire: write CTRL[0]=1 */
         kw_write(KALMAN_CTRL, 1.0);   /* bit[0]=1; double bit-pattern 0x3FF0...
      sets bit0=1 via lo word */
         /* Actually bit [0]=1 in the integer sense: write uint32 1 to lo word 
     of CTRL */
         /* Override: write via uint32 directly for CTRL (bit field, not 
     IEEE-754) */
         KALMAN_BASE[0] = 1U;   /* CTRL lo-word = 1 (start bit) */
         KALMAN_BASE[1] = 0U;   /* CTRL hi-word = 0 */

         /* 5. Poll STAT[1] (busy) — clear means done */
         uint32_t timeout = 100000U;
         while ((KALMAN_BASE[KALMAN_STAT >> 2] & 0x2U) && --timeout)
             ;
         if (!timeout) return -1;

         /* 6. Read x_out[0:2] at 0x30, 0x38, 0x40 */
         x_out[0] = kw_read(KALMAN_XOUT0);
         x_out[1] = kw_read(KALMAN_XOUT1);
         x_out[2] = kw_read(KALMAN_XOUT2);

         /* 7. Read P_out[0:8] at 0xA0..0xE0 */
         for (int i = 0; i < 9; i++)
             x_out[i] = kw_read(KALMAN_POUT0 + (uint32_t)(i * 8));
         /* Oops — must write to P_out not x_out. Use local indexing: */
         for (int i = 0; i < 9; i++)
             P_out[i] = kw_read(KALMAN_POUT0 + (uint32_t)(i * 8));

         return 0;
     }

     #endif /* KALMAN_HW_H */

     Note on CTRL write: CTRL is a bit-field register, not IEEE-754 double.
     Bypass kw_write
     and write raw uint32_t: KALMAN_BASE[0] = 1U; KALMAN_BASE[1] = 0U; to set
     start bit.

     Note on P_out loop: The inline above has a bug (writes to x_out twice). The
      actual
     implementation must use P_out for the second loop. Write the final file
     carefully.
